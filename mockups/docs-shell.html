Usage

Import the stylesheet once at your application root.

avatar-example.tsx 
import "@dixy-ui/react/styles.css";
import { Avatar, Theme } from "@dixy-ui/react";

export function Profile() {
  return (
    <Theme skin="dixy">
      <Avatar.Root size="md">
        <Avatar.Image src={url} alt="Akshay Saini" />
        <Avatar.Fallback>AS</Avatar.Fallback>
      </Avatar.Root>
    </Theme>
  );
}
